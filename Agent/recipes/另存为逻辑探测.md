只读探测 PS 内部的「组件另存为（新原型）」逻辑类 `CApSaveComponentAsNewPrototypeLogic`：确认它在当前进程里能否取到、成员形状、`Status` 枚举取值，以及当前选中的组件。

**不会调用 `Execute()`**，不改变场景。用于判断能否用 `InvokeUi=false` 无界面地走完「新建原型 + 新建组件」，从而绕开中文文件名校验。

## 参数

```json
[
  {"Name":"type_name","Label":"类型全名","Kind":"text","Required":false,"Default":"DnProcessSimulateCommands.SaveComponentAs.CApSaveComponentAsNewPrototypeLogic"}
]
```

## 执行按钮

```json
[
  {
    "id": "probe",
    "label": "只读探测",
    "args": {}
  }
]
```

## 代码

```csharp
var sb = new StringBuilder();

string want = Convert.ToString(type_name);
if (want == null) want = "";
want = want.Trim();
if (want.Length == 0)
    want = "DnProcessSimulateCommands.SaveComponentAs.CApSaveComponentAsNewPrototypeLogic";

sb.AppendLine("## 另存为逻辑探测");
sb.AppendLine();
sb.AppendLine("- 目标类型：`" + want + "`");
sb.AppendLine("- 模式：只读（不调用 Execute）");
sb.AppendLine();

// ── 1. 在已加载程序集里找 ──
System.Reflection.Assembly asm = null;
var loaded = new List<string>();
foreach (System.Reflection.Assembly a in AppDomain.CurrentDomain.GetAssemblies())
{
    string an = "";
    try { an = a.GetName().Name; } catch (Exception) { }
    if (an == "DnProcessSimulateCommands") asm = a;
    if (an.Length > 0 && an.StartsWith("Dn", StringComparison.OrdinalIgnoreCase)) loaded.Add(an);
}

string loadNote = asm == null ? "" : "已在 AppDomain 中找到";
if (asm == null)
{
    try
    {
        asm = System.Reflection.Assembly.Load("DnProcessSimulateCommands");
        loadNote = "已按名称加载";
    }
    catch (Exception ex)
    {
        loadNote = "加载失败：" + ex.GetType().Name + "：" + ex.Message;
    }
}

string asmName = "-";
string asmLoc = "-";
if (asm != null)
{
    try { asmName = asm.FullName; } catch (Exception) { asmName = "(读取失败)"; }
    try
    {
        string l = asm.Location;
        asmLoc = string.IsNullOrEmpty(l) ? "(无文件位置)" : l;
    }
    catch (Exception) { asmLoc = "(读取失败)"; }
}

sb.AppendLine("| 项目 | 值 |");
sb.AppendLine("| --- | --- |");
sb.AppendLine("| 程序集 | " + (asm == null ? "**未找到**" : "`" + asmName + "`") + " |");
sb.AppendLine("| 程序集位置 | " + (asmLoc.Length > 0 ? "`" + asmLoc + "`" : "-") + " |");
sb.AppendLine("| 加载情况 | " + loadNote + " |");
sb.AppendLine();

if (loaded.Count > 0)
{
    sb.AppendLine("已加载的 Dn* 程序集：" + string.Join("、", loaded.ToArray()));
    sb.AppendLine();
}

if (asm == null)
{
    sb.AppendLine("没有该程序集，无法继续。可能 PS 版本路径不同，或命令程序集按需加载且当前未触发。");
    return sb.ToString();
}

// ── 2. 取类型 ──
Type t = null;
try { t = asm.GetType(want, false); }
catch (Exception ex) { sb.AppendLine("GetType 异常：" + ex.Message); }

Type[] all = null;
try { all = asm.GetTypes(); }
catch (System.Reflection.ReflectionTypeLoadException ex) { all = ex.Types; }
catch (Exception) { }

if (t == null)
{
    sb.AppendLine("**类型未找到**。该程序集内与 SaveComponent 相关的类型：");
    sb.AppendLine();
    int shown = 0;
    if (all != null)
    {
        foreach (Type x in all)
        {
            if (x == null) continue;
            string fn = "";
            try { fn = x.FullName; } catch (Exception) { fn = x.Name; }
            if (fn != null && fn.IndexOf("SaveComponent", StringComparison.OrdinalIgnoreCase) >= 0)
            {
                sb.AppendLine("- `" + fn + "`");
                shown++;
            }
        }
    }
    if (shown == 0) sb.AppendLine("- （没有匹配项）");
    return sb.ToString();
}

string baseName = "-";
try { baseName = t.BaseType == null ? "-" : t.BaseType.FullName; } catch (Exception) { }

sb.AppendLine("| 项目 | 值 |");
sb.AppendLine("| --- | --- |");
sb.AppendLine("| 类型 | `" + t.FullName + "` |");
sb.AppendLine("| 可见性 | " + (t.IsPublic ? "public" : "non-public") + " |");
sb.AppendLine("| 基类 | " + baseName + " |");
sb.AppendLine();

var flags = System.Reflection.BindingFlags.Public | System.Reflection.BindingFlags.NonPublic
          | System.Reflection.BindingFlags.Instance | System.Reflection.BindingFlags.Static
          | System.Reflection.BindingFlags.DeclaredOnly;

sb.AppendLine("### 方法");
sb.AppendLine();
try
{
    foreach (System.Reflection.MethodInfo m in t.GetMethods(flags))
    {
        if (m.IsSpecialName) continue;
        string ps = "?";
        string rt = "?";
        try { rt = m.ReturnType == null ? "?" : m.ReturnType.Name; } catch (Exception) { }
        try
        {
            var arr = new List<string>();
            foreach (System.Reflection.ParameterInfo pi in m.GetParameters())
            {
                string ptn = "?";
                try { ptn = pi.ParameterType == null ? "?" : pi.ParameterType.Name; } catch (Exception) { }
                arr.Add(ptn + " " + pi.Name);
            }
            ps = string.Join(", ", arr.ToArray());
        }
        catch (Exception) { }
        sb.AppendLine("- `" + (m.IsPublic ? "" : "private ") + rt + " " + m.Name + "(" + ps + ")`");
    }
}
catch (Exception ex) { sb.AppendLine("- 枚举失败：" + ex.GetType().Name); }
sb.AppendLine();

sb.AppendLine("### 属性");
sb.AppendLine();
try
{
    foreach (System.Reflection.PropertyInfo pi in t.GetProperties(flags))
    {
        bool canRead = false, canWrite = false;
        try { canRead = pi.GetGetMethod(true) != null; } catch (Exception) { }
        try { canWrite = pi.GetSetMethod(true) != null; } catch (Exception) { }
        string ty = "?";
        try { ty = pi.PropertyType.Name; } catch (Exception) { }
        sb.AppendLine("- `" + ty + " " + pi.Name + "`（" + (canRead ? "可读" : "") + (canWrite ? "/可写" : "") + "）");
    }
}
catch (Exception ex) { sb.AppendLine("- 枚举失败：" + ex.GetType().Name); }
sb.AppendLine();

sb.AppendLine("### 字段");
sb.AppendLine();
try
{
    foreach (System.Reflection.FieldInfo fi in t.GetFields(flags))
    {
        string ty = "?";
        try { ty = fi.FieldType.Name; } catch (Exception) { }
        sb.AppendLine("- `" + (fi.IsPublic ? "" : "private ") + ty + " " + fi.Name + "`");
    }
}
catch (Exception ex) { sb.AppendLine("- 枚举失败：" + ex.GetType().Name); }
sb.AppendLine();

// ── 3. Status 枚举 ──
try
{
    Type st = t.GetNestedType("Status", System.Reflection.BindingFlags.Public | System.Reflection.BindingFlags.NonPublic);
    if (st != null)
    {
        sb.AppendLine("### Status 取值（失败原因清单）");
        sb.AppendLine();
        foreach (string name in Enum.GetNames(st)) sb.AppendLine("- `" + name + "`");
        sb.AppendLine();
    }
}
catch (Exception) { }

// ── 4. 当前选择 ──
sb.AppendLine("### 当前选择");
sb.AppendLine();
ITxComponent picked = null;
try
{
    TxObjectList sel = TxApplication.ActiveSelection.GetItems();
    if (sel != null)
    {
        foreach (ITxObject o in sel)
        {
            ITxComponent c = o as ITxComponent;
            if (c != null) { picked = c; break; }
        }
    }
}
catch (Exception ex) { sb.AppendLine("- 读取选择失败：" + ex.Message); }

if (picked == null)
{
    sb.AppendLine("没有选中任何组件。该逻辑很可能依赖当前选择集，正式调用前需要先选中目标组件。");
}
else
{
    string nm = "";
    string ty2 = "";
    bool open = false;
    try { nm = picked.Name == null ? "" : picked.Name; } catch (Exception) { }
    try { ty2 = picked.GetType().Name; } catch (Exception) { }
    try { open = picked.IsOpenForModeling; } catch (Exception) { }
    sb.AppendLine("- 组件：`" + nm + "`（" + ty2 + "）");
    sb.AppendLine("- 是否建模中：" + (open ? "是" : "否"));
    sb.AppendLine();
    sb.AppendLine("如果成员列表里没有显式的“组件”参数，基本可以确定 `Execute()` 读的就是这个选择集。");
}
sb.AppendLine();

sb.AppendLine("---");
sb.AppendLine();
sb.AppendLine("下一步：类型可取、成员形状符合预期时，再用 `InvokeUi=false` 调 `Execute()`，");
sb.AppendLine("目标文件名用 ASCII、目录选中文子目录，看它能否生成新组件并直接落到目标位置。");

return sb.ToString();
```
