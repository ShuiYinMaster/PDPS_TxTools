诊断组件的「结束建模」为什么失败：先只读打印存储状态与组件内容，确认无误后可按策略实际调用 `ITxComponent.EndModeling`。用于区分路径编码、文件名、组件名、原型/TuneData 这几类原因，不做路径搬运、不删除文件。

## 参数

```json
[
  {"Name":"targets","Label":"诊断组件","Kind":"objects","Required":true},
  {"Name":"end_target","Label":"手动指定目标路径","Kind":"text","Required":false,"Default":""},
  {"Name":"do_end","Label":"执行结束建模","Kind":"bool","Required":false,"Default":"false"},
  {"Name":"scan_scene","Label":"扫描场景基线","Kind":"bool","Required":false,"Default":"true"}
]
```

## 执行按钮

```json
[
  {
    "id": "scan",
    "label": "只诊断",
    "args": {
      "do_end": "false"
    }
  },
  {
    "id": "end",
    "label": "诊断并结束建模",
    "args": {
      "do_end": "true"
    }
  }
]
```

## 代码

```csharp
// 结束建模诊断 —— 自包含实现，不依赖插件中新增的诊断模块。
// 只读诊断不写盘；「诊断并结束建模」会实际调用 EndModeling，成功后组件存储位置改变。
var sb = new StringBuilder();
bool doEnd = Convert.ToBoolean(do_end);
bool scanScene = Convert.ToBoolean(scan_scene);
string manual = end_target == null ? "" : Convert.ToString(end_target).Trim();
string sysroot = "";
try { string root = TxApplication.SystemRootDirectory; sysroot = root == null ? "" : root; }
catch (Exception) { }

Func<string, bool> isAscii = delegate(string s)
{
    if (s == null) return true;
    for (int i = 0; i < s.Length; i++) if (s[i] > 127) return false;
    return true;
};

Func<ITxObject, string> nameOf = delegate(ITxObject o)
{
    if (o == null) return "";
    try { return o.Name == null ? "" : o.Name; }
    catch (Exception) { return ""; }
};

// ASCII 文件名：保留原名中的英文数字，附 8 位稳定哈希，避免不同组件撞名。
Func<string, string> asciiLeaf = delegate(string n)
{
    string src = n == null ? "" : n;
    var b = new StringBuilder();
    for (int i = 0; i < src.Length; i++)
    {
        char c = src[i];
        if (c < 128 && (char.IsLetterOrDigit(c) || c == '_' || c == '-')) b.Append(c);
        if (b.Length >= 24) break;
    }
    if (b.Length == 0) b.Append("EMD");
    uint h = 2166136261;
    for (int i = 0; i < src.Length; i++) { h = (h ^ src[i]) * 16777619; }
    return b.ToString() + "_" + h.ToString("x8") + ".cojt";
};

// 存储路径可能形如 "#\子目录\X.cojt"，按库根解析。
Func<string, string> resolve = delegate(string raw)
{
    if (raw == null || raw.Trim().Length == 0) return "";
    try
    {
        string p = raw.Replace('/', '\\');
        if (p.StartsWith("#\\"))
        {
            if (sysroot.Length == 0) return "";
            p = System.IO.Path.Combine(sysroot, p.Substring(2));
        }
        else if (!System.IO.Path.IsPathRooted(p))
        {
            if (sysroot.Length == 0) return "";
            p = System.IO.Path.Combine(sysroot, p);
        }
        return System.IO.Path.GetFullPath(p);
    }
    catch (Exception) { return ""; }
};

Func<string, bool> ensureDir = delegate(string dir)
{
    try
    {
        if (!System.IO.Directory.Exists(dir)) System.IO.Directory.CreateDirectory(dir);
        string probe = System.IO.Path.Combine(dir, ".emd-" + Guid.NewGuid().ToString("N"));
        using (var fs = new System.IO.FileStream(probe, System.IO.FileMode.CreateNew,
                   System.IO.FileAccess.Write, System.IO.FileShare.None, 1, System.IO.FileOptions.DeleteOnClose)) { }
        return true;
    }
    catch (Exception) { return false; }
};

// 目录清单：用于对比「EngData 临时 cojt」与「库内正式 cojt」的结构差异。
Func<string, string> listDir = delegate(string dir)
{
    if (dir == null || dir.Length == 0) return "（无路径）";
    try
    {
        if (!System.IO.Directory.Exists(dir)) return "（目录不存在）";
        var b = new StringBuilder();
        foreach (string f in System.IO.Directory.GetFiles(dir))
        {
            var fi = new System.IO.FileInfo(f);
            b.Append("`" + fi.Name + "`(" + (fi.Length / 1024) + "KB) ");
        }
        foreach (string d in System.IO.Directory.GetDirectories(dir))
            b.Append("[`" + System.IO.Path.GetFileName(d) + "`] ");
        string s = b.ToString().Trim();
        return s.Length == 0 ? "（空目录）" : s;
    }
    catch (Exception ex) { return "读取失败：" + ex.GetType().Name; }
};

Func<ITxComponent, bool> isOpen = delegate(ITxComponent c)
{
    try { return c.IsOpenForModeling; }
    catch (Exception) { return false; }
};

sb.AppendLine("## 结束建模诊断");
sb.AppendLine();
sb.AppendLine("- 库根：`" + (sysroot.Length == 0 ? "（空）" : sysroot) + "`（" + (isAscii(sysroot) ? "全 ASCII" : "含非 ASCII") + "）");
sb.AppendLine("- 模式：" + (doEnd ? "诊断并结束建模（会写盘）" : "只诊断（不写盘）"));
sb.AppendLine("- 调用：`ITxComponent.EndModeling(path)`，不经过 GUI 的 Save Component As 校验");
sb.AppendLine();

// ── 场景基线：用于判断「是这个组件特殊，还是整个场景都这样」 ──
if (scanScene)
{
    int visited = 0, libCount = 0, localCount = 0, otherCount = 0, openCount = 0;
    var libSamples = new List<string>();
    var stack = new Stack<ITxObject>();
    try
    {
        TxPhysicalRoot root = TxApplication.ActiveDocument.PhysicalRoot;
        if (root != null) stack.Push(root);
    }
    catch (Exception) { }

    while (stack.Count > 0 && visited < 3000)
    {
        ITxObject node = stack.Pop();
        visited++;

        ITxComponent nodeComp = node as ITxComponent;
        if (nodeComp != null)
        {
            string kind = "";
            try
            {
                ITxStorable st = nodeComp as ITxStorable;
                if (st != null && st.StorageObject != null) kind = st.StorageObject.GetType().Name;
            }
            catch (Exception) { }

            if (kind == "TxLibraryStorage")
            {
                libCount++;
                if (libSamples.Count < 5) libSamples.Add(nameOf(node));
            }
            else if (kind == "TxLocalStorage") localCount++;
            else otherCount++;

            try { if (nodeComp.IsOpenForModeling) openCount++; }
            catch (Exception) { }
        }

        ITxObjectCollection coll = node as ITxObjectCollection;
        if (coll == null) continue;
        try
        {
            TxObjectList kids = coll.GetDirectDescendants(new TxTypeFilter(typeof(ITxObject)));
            if (kids == null) continue;
            foreach (ITxObject kid in kids)
                if (kid != null && stack.Count < 5000) stack.Push(kid);
        }
        catch (Exception) { }
    }

    sb.AppendLine("### 场景基线");
    sb.AppendLine();
    sb.AppendLine("扫描 " + visited + " 个对象。组件存储类型分布：");
    sb.AppendLine();
    sb.AppendLine("| 存储类型 | 数量 | 说明 |");
    sb.AppendLine("| --- | ---: | --- |");
    sb.AppendLine("| TxLibraryStorage | " + libCount + " | 已入库，有正式库引用 |");
    sb.AppendLine("| TxLocalStorage | " + localCount + " | 本地建模，从未入库 |");
    sb.AppendLine("| 其他 / 无存储 | " + otherCount + " | 容器或其他类型 |");
    sb.AppendLine("| 仍在建模中 | " + openCount + " | 需确认是否卡住 |");
    sb.AppendLine();
    if (libSamples.Count > 0)
    {
        sb.AppendLine("可选对照组（TxLibraryStorage）：");
        for (int i = 0; i < libSamples.Count; i++) sb.AppendLine("- " + libSamples[i]);
        sb.AppendLine();
        sb.AppendLine("建议对这些组件单独跑一次「只诊断」：若它们的存储类型为 TxLibraryStorage 且");
        sb.AppendLine("`A 原样` 策略能成功结束建模，说明 API 正常，问题只在当前这个 TxLocalStorage 组件上。");
        sb.AppendLine();
    }
    else
    {
        sb.AppendLine("场景里没有任何 TxLibraryStorage 组件，无法做基线对照。");
        sb.AppendLine();
    }
}

int total = 0, handled = 0, succeeded = 0;

foreach (ITxObject raw in targets)
{
    total++;
    ITxComponent comp = raw as ITxComponent;
    if (comp == null)
    {
        log("[跳过] " + nameOf(raw) + "：不是 ITxComponent，无法建模。");
        continue;
    }
    handled++;

    string name = nameOf(comp);
    string storageKind = "无存储对象";
    string rawStorage = null;
    try
    {
        ITxStorable storable = comp as ITxStorable;
        if (storable != null && storable.StorageObject != null)
        {
            storageKind = storable.StorageObject.GetType().Name;
            TxLibraryStorage lib = storable.StorageObject as TxLibraryStorage;
            if (lib != null) rawStorage = lib.FullPath;
        }
    }
    catch (Exception ex) { storageKind = "读取失败：" + ex.GetType().Name; }

    string resolved = resolve(rawStorage);
    string currentName = name;
    string leaf = "";
    string dir = "";
    if (resolved.Length > 0)
    {
        try
        {
            leaf = System.IO.Path.GetFileName(resolved);
            dir = System.IO.Path.GetDirectoryName(resolved);
        }
        catch (Exception) { }
    }

    var kids = new List<ITxObject>();
    try
    {
        ITxObjectCollection coll = comp as ITxObjectCollection;
        if (coll != null)
        {
            TxObjectList list = coll.GetDirectDescendants(new TxTypeFilter(typeof(ITxObject)));
            if (list != null) { foreach (ITxObject kid in list) if (kid != null) kids.Add(kid); }
        }
    }
    catch (Exception) { }

    bool referenced = false;
    try { referenced = comp.HasReferencedEntities; } catch (Exception) { }
    bool canOpen = false, opened = false;
    try { canOpen = comp.CanOpenForModeling; } catch (Exception) { }
    opened = isOpen(comp);

    sb.AppendLine("### " + (name.Length == 0 ? "(未命名组件)" : name));
    sb.AppendLine();
    sb.AppendLine("| 项目 | 值 |");
    sb.AppendLine("| --- | --- |");
    sb.AppendLine("| SDK 类型 | " + comp.GetType().Name + " |");
    sb.AppendLine("| 可建模 / 建模中 | " + (canOpen ? "是" : "否") + " / " + (opened ? "是" : "否") + " |");
    sb.AppendLine("| HasReferencedEntities | " + (referenced ? "是" : "否") + " |");
    sb.AppendLine("| 直接子对象数 | " + kids.Count + " |");
    sb.AppendLine("| 存储类型 | " + storageKind + " |");
    sb.AppendLine("| 原始引用 | " + (rawStorage == null || rawStorage.Length == 0 ? "（空）" : "`" + rawStorage + "`") + " |");
    sb.AppendLine("| 解析路径 | " + (resolved.Length == 0 ? "（无法解析）" : "`" + resolved + "`") + " |");
    if (resolved.Length > 0)
        sb.AppendLine("| 库内 cojt 内容 | " + listDir(resolved) + " |");

    // 原型 / 类型定义探测：只读。缺少原型是 EndModeling 失败的常见原因之一。
    string[] probes = new string[] { "Prototype", "PlanningRepresentation", "ComponentType", "EquipmentType", "PrototypeType" };
    for (int pi = 0; pi < probes.Length; pi++)
    {
        var prop = comp.GetType().GetProperty(probes[pi]);
        if (prop == null) continue;
        string val;
        try
        {
            object rawValue = prop.GetValue(comp, null);
            if (rawValue == null) { val = "null"; }
            else
            {
                val = Convert.ToString(rawValue);
                if (val == null || val.Length == 0) val = rawValue.GetType().Name;
            }
        }
        catch (Exception ex) { val = "读取失败：" + ex.GetType().Name; }
        sb.AppendLine("| " + probes[pi] + " | " + val + " |");
    }
    sb.AppendLine();

    if (kids.Count > 0)
    {
        sb.AppendLine("子对象示例：" + nameOf(kids[0]) + "（" + kids[0].GetType().Name + "）");
        if (kids.Count > 1) sb.AppendLine("、 " + nameOf(kids[1]) + "（" + kids[1].GetType().Name + "）");
        sb.AppendLine();
    }

    // ── 构造策略 ──
    var labels = new List<string>();
    var paths = new List<string>();

    if (manual.Length > 0)
    {
        labels.Add("M 手动指定路径");
        paths.Add(manual);
    }
    else
    {
        if (resolved.Length > 0)
        {
            labels.Add("A 原样（原路径＋原文件名）");
            paths.Add(resolved);
        }
        if (resolved.Length == 0 && sysroot.Length > 0 && name.Length > 0)
        {
            labels.Add("B 库根＋原组件名");
            paths.Add(System.IO.Path.Combine(sysroot,
                name.EndsWith(".cojt", StringComparison.OrdinalIgnoreCase) ? name : name + ".cojt"));
        }
        if (dir != null && dir.Length > 0 && !isAscii(leaf))
        {
            labels.Add("C 原目录＋ASCII 文件名");
            paths.Add(System.IO.Path.Combine(dir, asciiLeaf(name)));
        }
        if (sysroot.Length > 0)
        {
            labels.Add("F 库根内 ASCII 子目录＋ASCII 文件名");
            paths.Add(System.IO.Path.Combine(sysroot, "_TxToolsEMD", asciiLeaf(name)));
        }
        if (!isAscii(name) && sysroot.Length > 0)
        {
            labels.Add("E 临时 ASCII 组件名＋库根内 ASCII 路径");
            paths.Add(System.IO.Path.Combine(sysroot, "_TxToolsEMD", asciiLeaf(name)));
        }
        labels.Add("D 库根之外的纯 ASCII 目录");
        paths.Add(System.IO.Path.Combine(System.IO.Path.GetTempPath(), "TxToolsEndModeling", asciiLeaf(name)));
    }

    sb.AppendLine("候选策略：");
    sb.AppendLine();
    for (int i = 0; i < labels.Count; i++)
        sb.AppendLine((i + 1) + ". **" + labels[i] + "** → `" + paths[i] + "`");
    sb.AppendLine();

    if (!doEnd)
    {
        log("[只诊断] " + name + "：存储=" + storageKind + "，子对象=" + kids.Count + "，候选策略=" + labels.Count);
        continue;
    }

    // ── 执行 ──
    if (!opened)
    {
        if (!canOpen)
        {
            sb.AppendLine("**未执行**：该组件当前不允许打开建模（CanOpenForModeling=false）。");
            sb.AppendLine();
            continue;
        }
        try
        {
            comp.SetModelingScope();
            sb.AppendLine("已调用 `SetModelingScope` 打开建模（不会自动收回）。");
            sb.AppendLine();
        }
        catch (Exception ex)
        {
            sb.AppendLine("**未执行**：打开建模失败 — " + ex.GetType().Name + "：" + ex.Message);
            sb.AppendLine();
            continue;
        }
    }

    sb.AppendLine("尝试结果：");
    sb.AppendLine();

    // 本地存储探针：TxLocalStorage 的建模数据如果已经写不出去，
    // 后面的 EndModeling 必然失败，且错误会退化成同一句通用文案。
    // SaveComponent() 会保存建模改动到本地 JT 并返回路径 —— 成功当且仅当本地存储可写。
    if (storageKind == "TxLocalStorage")
    {
        try
        {
            ITxStorable probeStorable = comp as ITxStorable;
            TxLocalStorage local = probeStorable == null ? null : probeStorable.StorageObject as TxLocalStorage;
            if (local == null)
            {
                sb.AppendLine("- 本地存储探针：StorageObject 不是 TxLocalStorage，跳过。");
            }
            else
            {
                string jtPath = local.SaveComponent();
                sb.AppendLine("- 本地存储探针：`SaveComponent()` → "
                    + (jtPath == null || jtPath.Length == 0 ? "（返回空路径）" : "`" + jtPath + "`"));
                if (jtPath != null && jtPath.Length > 0)
                {
                    string jtDir = "";
                    try { jtDir = System.IO.Path.GetDirectoryName(jtPath); } catch (Exception) { }
                    sb.AppendLine("  - 临时 cojt 内容：" + listDir(jtDir));
                    sb.AppendLine("  - 本地 JT 可写；失败点在其后的「转成正式 cojt / 写回库」环节。");
                }
                else
                {
                    sb.AppendLine("  - 返回空路径：本地建模数据可能没有落盘位置，这是 EndModeling 失败的直接嫌疑。");
                }
            }
        }
        catch (Exception ex)
        {
            sb.AppendLine("- 本地存储探针：`SaveComponent()` 失败 — " + ex.GetType().Name + "：" + ex.Message);
            sb.AppendLine("  - 本地存储本身写不出去；EndModeling 之前的建模数据已经不完整。");
        }
        sb.AppendLine();
    }

    string successPath = null;

    for (int k = 0; k < labels.Count; k++)
    {
        if (!isOpen(comp)) break;

        string label = labels[k];
        string target = paths[k];
        string parent = "";
        try { parent = System.IO.Path.GetDirectoryName(target); } catch (Exception) { }
        if (parent == null || parent.Length == 0)
        {
            sb.AppendLine("- **" + label + "**：目标父目录无效 → `" + target + "`");
            continue;
        }
        if (!ensureDir(parent))
        {
            sb.AppendLine("- **" + label + "**：父目录不可创建或不可写 → `" + parent + "`");
            continue;
        }

        bool renamed = false;
        string originalName = null;
        if (label.StartsWith("E", StringComparison.Ordinal))
        {
            try
            {
                originalName = nameOf(comp);
                ((ITxObject)comp).Name = "EMD_" + Guid.NewGuid().ToString("N").Substring(0, 8);
                renamed = true;
            }
            catch (Exception ex)
            {
                sb.AppendLine("- **" + label + "**：临时改名失败 — " + ex.Message);
                continue;
            }
        }

        try
        {
            comp.EndModeling(target);
            successPath = target;
            succeeded++;
            sb.AppendLine("- **" + label + "**：✅ 成功 → `" + target + "`");
            log("[成功] " + name + "：" + label);
        }
        catch (Exception ex)
        {
            sb.AppendLine("- **" + label + "**：❌ " + ex.GetType().Name + "：" + ex.Message);
            log("[失败] " + name + "：" + label + " — " + ex.Message);
        }
        finally
        {
            if (renamed)
            {
                try
                {
                    ((ITxObject)comp).Name = originalName;
                    sb.AppendLine("  - 已恢复原名：" + originalName);
                }
                catch (Exception ex)
                {
                    sb.AppendLine("  - ⚠ 恢复原名失败：" + ex.Message + "；当前名称：" + nameOf(comp));
                }
            }
        }

        if (successPath != null) break;
    }

    sb.AppendLine();
    if (successPath != null)
        sb.AppendLine("**结论**：可用路径为 `" + successPath + "`。");
    else if (isOpen(comp))
        sb.AppendLine("**结论**：全部策略失败，组件仍处于建模状态。失败与路径无关时，重点看原型（Define Component Type）、TuneData 与组件内容。");
    else
        sb.AppendLine("**结论**：组件建模已结束，但未记录到成功路径，请核对场景实际状态。");
    sb.AppendLine();
}

if (total == 0) throw new Exception("没有绑定任何对象，请先在“诊断组件”里选取组件。");
if (handled == 0) throw new Exception("绑定的对象里没有 ITxComponent，请选取资源/组件实例而不是几何体或操作。");

sb.AppendLine("---");
sb.AppendLine();
sb.AppendLine("共 " + total + " 个对象，可诊断 " + handled + " 个" + (doEnd ? "，成功 " + succeeded + " 个" : "（只诊断，未写盘）") + "。");
if (doEnd)
{
    sb.AppendLine();
    sb.AppendLine("提示：`_TxToolsEMD` 是诊断专用目录，确认结论后可手工删除；成功结束建模会改变组件存储位置。");
}

return sb.ToString();
```
