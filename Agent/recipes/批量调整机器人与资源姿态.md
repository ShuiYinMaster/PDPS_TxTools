将当前工程中的机器人和资源设备批量切换到各自已保存的同名姿态。基于原始“所有机器人回 HOME”脚本扩展，适用于机器人、焊枪、夹具等具有运动学和命名姿态的设备。

### 使用方法

1. 在配方栏点击“导入”，选择此 Markdown 文件。
2. **姿态名称可选填**：默认 `HOME`；可改为 `OPEN`、`CLOSE` 或设备中已有的其他姿态名称。清空输入时仍使用 `HOME`。名称按设备保存的名称精确匹配。
3. **调整范围可选**：不绑定对象时，遍历当前工程；要限定范围，在 PS 中选中机器人、资源设备或其父级资源组，再点击“取选择”。选中的设备本身及所选组下的设备都会纳入，重叠范围自动去重。PS 的临时选中状态不会自动成为执行范围，以配方栏的绑定为准。
4. 默认同时调整机器人和资源设备。取消相应勾选即可只调整其中一类；“资源设备”指机器人以外、支持已保存姿态的运动学设备。
5. 点击“执行”。逐项显示成功、缺少姿态和失败信息，最后汇总数量。找不到目标姿态时保留该设备当前姿态，并列出它已有的姿态名称供核对。

### 执行效果与范围

- 使用设备自己的 `GetPoseByName` 和 `CurrentPose`，调整其关节姿态，不改变设备在场景中的整体安装坐标，不创建或重命名姿态。
- 静态底座、围栏、普通零件等不具备设备命名姿态的对象不参与调整；资源组仅作为设备遍历范围。
- 单个设备失败不阻断其余设备，日志会保留失败原因；遍历异常也会单独统计。
- 姿态名称是一个可修改的文本参数，可填写任意已保存的名称。多个设备各自缺少该名称时会跳过，不会替换为别的姿态。
- 导入只保存配方，不执行脚本。执行时使用当前工程对象，不含原工程的对象 ID。

## 代码

```csharp
// 批量调整机器人与资源设备的命名关节姿态。
// 参数变量由配方运行器注入；这里必须使用方法体，不能保留 MainScript 类外壳。
var doc = TxApplication.ActiveDocument;
if (doc == null) throw new Exception("当前没有打开的 Study。");

string requestedPose = Convert.ToString(pose_name).Trim();
if (requestedPose.Length == 0) requestedPose = "HOME";
bool adjustRobots = Convert.ToBoolean(include_robots);
bool adjustResources = Convert.ToBoolean(include_resources);
if (!adjustRobots && !adjustResources)
    throw new Exception("请至少勾选“调整机器人”或“调整资源设备”中的一项。");

var devices = new List<ITxDevice>();
var foundIds = new HashSet<string>(StringComparer.Ordinal);
int traversalFailed = 0;

Action<object> addDevice = delegate(object item)
{
    var device = item as ITxDevice;
    if (device == null) return;
    bool isRobot = device is ITxRobot;
    if ((isRobot && !adjustRobots) || (!isRobot && !adjustResources)) return;
    var txObject = item as ITxObject;
    if (txObject == null) return;
    string id = Convert.ToString(txObject.Id);
    if (string.IsNullOrEmpty(id)) throw new Exception("设备缺少有效对象 ID，无法去重。");
    if (foundIds.Add(id)) devices.Add(device);
};

Action<object, string> collect = delegate(object root, string rootLabel)
{
    if (root == null) return;
    try
    {
        // GetAllDescendants 不包含根本身，所以先纳入直接选中的设备。
        addDevice(root);
        var collection = root as ITxObjectCollection;
        if (collection == null)
        {
            if (!(root is ITxDevice)) log("[跳过范围] " + rootLabel + "：不是设备，也不是可遍历的对象组。");
            return;
        }
        var descendants = collection.GetAllDescendants(new TxTypeFilter(typeof(ITxDevice)));
        if (descendants == null) return;
        foreach (object item in descendants) addDevice(item);
    }
    catch (Exception ex)
    {
        traversalFailed++;
        log("[遍历失败] " + rootLabel + "：" + ex.Message);
    }
};

var explicitRoots = targets as System.Collections.IEnumerable;
int rootCount = 0;
if (explicitRoots != null)
{
    foreach (object root in explicitRoots)
    {
        rootCount++;
        string rootLabel = "所选对象 " + rootCount;
        try
        {
            var txObject = root as ITxObject;
            if (txObject != null) rootLabel = txObject.Name;
        }
        catch { }
        collect(root, rootLabel);
    }
}

if (rootCount == 0)
{
    collect(doc.PhysicalRoot, "物理根节点");
    // 保留原脚本对其他根节点的兼容；当前版本没有相应属性时不访问。
    foreach (string propertyName in new string[] { "ResourceRoot", "ComponentRoot" })
    {
        try
        {
            var property = doc.GetType().GetProperty(propertyName,
                System.Reflection.BindingFlags.Instance | System.Reflection.BindingFlags.Public);
            if (property != null) collect(property.GetValue(doc, null), propertyName);
        }
        catch (Exception ex)
        {
            traversalFailed++;
            log("[遍历失败] " + propertyName + "：" + ex.Message);
        }
    }
}

log("目标姿态：" + requestedPose + "；范围：" + (rootCount == 0 ? "当前工程" : "配方绑定对象及其下属设备")
    + "；找到 " + devices.Count + " 个待调整设备。");

int robotSuccess = 0;
int resourceSuccess = 0;
int noPose = 0;
int failed = 0;
foreach (ITxDevice device in devices)
{
    bool isRobot = device is ITxRobot;
    string category = isRobot ? "机器人" : "资源设备";
    string deviceName = "(未命名)";
    try
    {
        var txObject = device as ITxObject;
        if (txObject != null && !string.IsNullOrEmpty(txObject.Name)) deviceName = txObject.Name;
        TxPose namedPose = device.GetPoseByName(requestedPose);
        var poseData = namedPose == null ? null : namedPose.PoseData;
        if (poseData == null)
        {
            noPose++;
            var available = new List<string>();
            int availableCount = 0;
            try
            {
                var poseList = device.PoseList;
                if (poseList != null)
                {
                    foreach (object item in poseList)
                    {
                        var pose = item as TxPose;
                        if (pose == null) continue;
                        availableCount++;
                        if (available.Count < 8) available.Add(pose.Name);
                    }
                }
            }
            catch { available.Add("(无法读取姿态列表)"); }
            string existing = available.Count == 0 ? "无已保存姿态" : string.Join("、", available.ToArray());
            if (availableCount > 8) existing += " 等 " + availableCount + " 项";
            log("[跳过] " + category + " " + deviceName + "：未找到可用的 " + requestedPose
                + "。已有姿态：" + existing);
            continue;
        }
        device.CurrentPose = poseData;
        if (isRobot) robotSuccess++; else resourceSuccess++;
        log("[完成] " + category + " " + deviceName + " -> " + requestedPose);
    }
    catch (Exception ex)
    {
        failed++;
        log("[失败] " + category + " " + deviceName + "：" + ex.Message);
    }
}

string summary = "完成：成功 " + (robotSuccess + resourceSuccess) + " 个（机器人 " + robotSuccess
    + " 个，资源设备 " + resourceSuccess + " 个），未找到可用姿态 " + noPose
    + " 个，调整失败 " + failed + " 个，范围遍历异常 " + traversalFailed + " 处。";
if (devices.Count == 0) log("当前范围没有符合类别条件的运动学设备。请检查范围绑定和类别勾选。");
return summary;
```
