批量将机器人、焊枪、夹具等设备切换到已保存的指定姿态。

1. 选择设备或资源组。要调整整个工程，选择“当前研究全部设备”。
2. 填写姿态名称，留空使用 `HOME`；按需选择设备类别和是否包含下级设备。
3. 点击“执行”调整姿态。

没有指定姿态的设备会跳过。静态零件不参与调整。

## 代码

```csharp
var doc = TxApplication.ActiveDocument;
if (doc == null) throw new Exception("当前没有打开的研究。");
string requestedPose = (Convert.ToString(pose_name) ?? "").Trim();
if (requestedPose.Length == 0) requestedPose = "HOME";
string scope = Convert.ToString(scope_mode);
string kind = Convert.ToString(device_kind);
if (scope != "selected" && scope != "study") throw new Exception("无效的范围选项。");
if (kind != "both" && kind != "robots" && kind != "resources") throw new Exception("无效的设备类别。");
bool recursive = Convert.ToBoolean(include_children);
var devices = new List<ITxDevice>();
var found = new HashSet<string>(StringComparer.Ordinal);
Action<ITxDevice> addDevice = delegate(ITxDevice device)
{
    if (device == null) return;
    bool robot = device is ITxRobot;
    if ((robot && kind == "resources") || (!robot && kind == "robots")) return;
    if (found.Add(Convert.ToString(device.Id))) devices.Add(device);
};
Action<ITxObject, bool> collect = delegate(ITxObject root, bool expand)
{
    addDevice(root as ITxDevice);
    var collection = root as ITxObjectCollection;
    if (expand && collection != null)
        foreach (ITxObject child in collection.GetAllDescendants(new TxTypeFilter(typeof(ITxDevice))))
            addDevice(child as ITxDevice);
};
if (scope == "study")
{
    if (doc.PhysicalRoot == null) throw new Exception("当前研究没有物理根节点。");
    collect(doc.PhysicalRoot, true);
}
else
{
    var roots = targets as System.Collections.IEnumerable;
    int rootCount = 0;
    if (roots != null)
        foreach (ITxObject root in roots)
        {
            if (root == null) continue;
            rootCount++;
            if (root is ITxDevice) { collect(root, recursive); continue; }
            // 选择设备内部几何或链接时，只定位最近的所属设备。
            ITxObject ancestor = root.Collection as ITxObject;
            ITxDevice owner = null;
            var visited = new HashSet<string>(StringComparer.Ordinal);
            while (ancestor != null && visited.Add(Convert.ToString(ancestor.Id)))
            {
                owner = ancestor as ITxDevice;
                if (owner != null) break;
                ancestor = ancestor.Collection as ITxObject;
            }
            if (owner != null) { addDevice(owner); continue; }
            if (recursive && root is ITxObjectCollection) collect(root, true);
            else log("[跳过范围] " + root.Name + "：没有所属设备；可绑定设备或启用“包含下级设备”。");
        }
    if (rootCount == 0) throw new Exception("请先绑定设备或资源组；若要处理整个研究，请显式选择“当前研究全部设备”。");
}
devices = devices.OrderBy(d => d.Name, StringComparer.OrdinalIgnoreCase)
    .ThenBy(d => Convert.ToString(d.Id), StringComparer.Ordinal).ToList();
if (devices.Count == 0) throw new Exception("当前范围没有符合类别的运动学设备。静态几何不支持命名关节姿态。");
var ready = new List<ITxDevice>();
var desired = new List<TxPoseData>();
var original = new List<TxPoseData>();
int missing = 0, failed = 0, robotSuccess = 0, resourceSuccess = 0;
try
{
    // 全部候选完成读取和快照后再修改，读取异常不会留下部分执行结果。
    foreach (var device in devices)
    {
        var pose = device.GetPoseByName(requestedPose);
        var data = pose == null ? null : pose.PoseData;
        if (data == null)
        {
            missing++;
            var available = new List<string>();
            if (device.PoseList != null)
                foreach (object item in device.PoseList)
                {
                    var saved = item as TxPose;
                    if (saved != null) available.Add(saved.Name);
                }
            log("[缺少姿态跳过] " + device.Name + "：" + requestedPose + "；已有："
                + (available.Count == 0 ? "无" : string.Join("、", available.Take(8).ToArray()))
                + (available.Count > 8 ? " 等 " + available.Count + " 项" : ""));
            continue;
        }
        var current = device.CurrentPose;
        if (current == null) throw new Exception(device.Name + " 无法读取当前姿态，未执行调整。");
        desired.Add(new TxPoseData(data));
        original.Add(new TxPoseData(current));
        ready.Add(device);
        log("[候选] " + (device is ITxRobot ? "机器人 " : "资源设备 ") + device.Name + " -> " + requestedPose);
    }
    for (int i = 0; i < ready.Count; i++)
        {
            var device = ready[i];
            try
            {
                device.CurrentPose = desired[i];
                if (device is ITxRobot) robotSuccess++; else resourceSuccess++;
                log("[完成] " + device.Name + " -> " + requestedPose);
            }
            catch (Exception ex)
            {
                failed++;
                try { device.CurrentPose = original[i]; }
                catch (Exception restore)
                { throw new Exception(device.Name + " 调整失败且恢复失败，已停止后续调整，请撤销本次操作。调整：" + ex.Message + "；恢复：" + restore.Message); }
                log("[失败并已恢复] " + device.Name + "：" + ex.Message);
            }
        }
    if (failed > 0 && robotSuccess + resourceSuccess == 0)
        throw new Exception("没有成功调整设备，失败设备已恢复原姿态；详情见日志。");
    return "## 姿态调整结果"
        + "\n\n- 范围：" + (scope == "study" ? "当前研究全部设备" : "所选对象")
        + "\n- 目标姿态：" + requestedPose
        + "\n- 成功：" + (robotSuccess + resourceSuccess) + " 个（机器人 " + robotSuccess
        + "，资源设备 " + resourceSuccess + "）"
        + "\n- 缺少姿态跳过：" + missing + " 个"
        + "\n- 失败并恢复：" + failed + " 个";

}
finally
{
    foreach (var data in desired) data.Dispose();
    foreach (var data in original) data.Dispose();
}
```
