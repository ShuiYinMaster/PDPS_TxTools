对选中组件做「另存为（新原型）」的**只读预演**：反射调用内部逻辑的 `ValidateComponent` 与 `ComputeNewPath`，提前拿到校验结果、`Status` 失败原因和目标路径，**不调用 `Execute()`**，不创建、不修改任何对象。

用于判断直接走内部另存为是否可行，以及它会往哪个目录写。

## 参数

```json
[
  {"Name":"type_name","Label":"类型全名","Kind":"text","Required":false,"Default":"DnProcessSimulateCommands.SaveComponentAs.CApSaveComponentAsNewPrototypeLogic"},
  {"Name":"new_file_name","Label":"拟用文件名（留空=自动）","Kind":"text","Required":false,"Default":""}
]
```

## 执行按钮

```json
[
  {
    "id": "dryrun",
    "label": "只读预演",
    "args": {}
  }

```

var sb = new StringBuilder();
var bf = System.Reflection.BindingFlags.Public | System.Reflection.BindingFlags.NonPublic
       | System.Reflection.BindingFlags.Instance | System.Reflection.BindingFlags.Static;

string want = Convert.ToString(type_name);
if (want == null) want = "";
want = want.Trim();
if (want.Length == 0)
    want = "DnProcessSimulateCommands.SaveComponentAs.CApSaveComponentAsNewPrototypeLogic";

string fileNameWanted = Convert.ToString(new_file_name);
if (fileNameWanted == null) fileNameWanted = "";
fileNameWanted = fileNameWanted.Trim();

Func<string, bool> IsAsciiText = delegate(string s)
{
    if (s == null) return true;
    for (int i = 0; i < s.Length; i++) if (s[i] > 127) return false;
    return true;
};

Func<object, string> statusOf = delegate(object target)
{
    try
    {
        System.Reflection.FieldInfo f = target.GetType().GetField("_status", bf);
        if (f == null) return "(无 _status 字段)";
        object v = f.GetValue(target);
        if (v == null) return "(null)";
        string s = Convert.ToString(v);
        return s == null ? "(null)" : s;
    }
    catch (Exception ex) { return "(读取失败：" + ex.GetType().Name + ")"; }
};

Func<object, string, object[], object> call = delegate(object target, string name, object[] args)
{
    System.Reflection.MethodInfo m = target.GetType().GetMethod(name, bf);
    if (m == null) throw new MissingMethodException(name);
    try { return m.Invoke(target, args); }
    catch (System.Reflection.TargetInvocationException tie)
    {
        throw (tie.InnerException ?? tie);
    }
};

sb.AppendLine("## 另存为预演");
sb.AppendLine();
sb.AppendLine("- 类型：`" + want + "`");
sb.AppendLine("- 模式：只读（不调用 Execute）");
sb.AppendLine();

ITxComponent comp = null;
string compName = "";
try
{
    TxObjectList sel = TxApplication.ActiveSelection.GetItems();
    if (sel != null)
    {
        foreach (ITxObject o in sel)
        {
            ITxComponent c = o as ITxComponent;
            if (c != null) { comp = c; break; }
        }
    }
}
catch (Exception ex) { sb.AppendLine("读取选择失败：" + ex.Message); }

if (comp == null)
{
    sb.AppendLine("**没有选中组件**。请先在场景里选中目标组件再执行本预演。");
    return sb.ToString();
}

try { compName = comp.Name == null ? "" : comp.Name; } catch (Exception) { }

bool openModeling = false;
try { openModeling = comp.IsOpenForModeling; } catch (Exception) { }
string storageKind = "无";
string storagePath = "";
try
{
    ITxStorable st = comp as ITxStorable;
    if (st != null && st.StorageObject != null)
    {
        storageKind = st.StorageObject.GetType().Name;
        TxLibraryStorage lib = st.StorageObject as TxLibraryStorage;
        if (lib != null)
        {
            storagePath = lib.FullPath;
            if (storagePath != null) storagePath = storagePath.Replace('/', '\\');
        }
    }
}
catch (Exception) { }

sb.AppendLine("| 项目 | 值 |");
sb.AppendLine("| --- | --- |");
sb.AppendLine("| 组件 | `" + compName + "` |");
sb.AppendLine("| 是否建模中 | " + (openModeling ? "是" : "否") + " |");
sb.AppendLine("| 存储类型 | " + storageKind + " |");
sb.AppendLine("| 存储路径 | " + (storagePath.Length == 0 ? "（空）" : "`" + storagePath + "`") + " |");
sb.AppendLine();

// ── 找类型并实例化 ──
System.Reflection.Assembly asm = null;
foreach (System.Reflection.Assembly a in AppDomain.CurrentDomain.GetAssemblies())
{
    string an = "";
    try { an = a.GetName().Name; } catch (Exception) { }
    if (an == "DnProcessSimulateCommands") { asm = a; break; }
}
if (asm == null)
{
    sb.AppendLine("程序集 `DnProcessSimulateCommands` 未加载，无法继续。");
    return sb.ToString();
}

Type t = asm.GetType(want, false);
if (t == null)
{
    sb.AppendLine("类型未找到：`" + want + "`");
    return sb.ToString();
}

object inst = null;
try { inst = Activator.CreateInstance(t); }
catch (Exception ex)
{
    sb.AppendLine("实例化失败（可能没有无参构造）：" + ex.GetType().Name + "：" + ex.Message);
    return sb.ToString();
}

// ── 1. ValidateComponent ──
sb.AppendLine("### 校验结果");
sb.AppendLine();
string validateError = null;
object validateResult = null;
try
{
    validateResult = call(inst, "ValidateComponent", new object[] { comp });
}
catch (Exception ex)
{
    validateError = ex.GetType().Name + "：" + ex.Message;
}

if (validateError != null)
{
    sb.AppendLine("- **调用异常**：" + validateError);
}
else
{
    bool ok = validateResult is bool && (bool)validateResult;
    sb.AppendLine("- ValidateComponent 返回：**" + (ok ? "true（通过）" : "false（不通过）") + "**");
    sb.AppendLine("- 调用后 _status：" + statusOf(inst));
}
sb.AppendLine();

// ── 2. ComputeNewPath ──
sb.AppendLine("### 目标路径计算");
sb.AppendLine();
string computed = null;
string computeError = null;
try
{
    object r = call(inst, "ComputeNewPath", new object[] { comp });
    computed = r == null ? null : Convert.ToString(r);
}
catch (Exception ex)
{
    computeError = ex.GetType().Name + "：" + ex.Message;
}

if (computeError != null)
{
    sb.AppendLine("- **调用异常**：" + computeError);
}
else if (computed == null || computed.Length == 0)
{
    sb.AppendLine("- ComputeNewPath 返回空。");
}
else
{
    sb.AppendLine("- ComputeNewPath：" + "`" + computed + "`");
    sb.AppendLine("- 路径全 ASCII：" + (IsAsciiText(computed) ? "是" : "**否**"));

    object under = null;
    string underError = null;
    try { under = call(inst, "IsPathUnderSysroot", new object[] { computed }); }
    catch (Exception ex) { underError = ex.GetType().Name + "：" + ex.Message; }
    if (underError != null) sb.AppendLine("- IsPathUnderSysroot 调用异常：" + underError);
    else sb.AppendLine("- IsPathUnderSysroot：" + ((under is bool && (bool)under) ? "**true**" : "**false**"));

    string finalFileName = fileNameWanted;
    if (finalFileName.Length == 0)
    {
        try
        {
            finalFileName = System.IO.Path.GetFileName(computed);
        }
        catch (Exception) { finalFileName = ""; }
    }
    if (finalFileName.Length > 0 && fileNameWanted.Length > 0)
    {
        object pick = null;
        string pickError = null;
        try { pick = call(inst, "PickNewPath", new object[] { computed, fileNameWanted }); }
        catch (Exception ex) { pickError = ex.GetType().Name + "：" + ex.Message; }
        if (pickError != null) sb.AppendLine("- PickNewPath 调用异常：" + pickError);
        else sb.AppendLine("- PickNewPath（拟用名 " + fileNameWanted + "）：`" + Convert.ToString(pick) + "`");
    }
}
sb.AppendLine();

// ── 3. 结论 ──
sb.AppendLine("### 判定");
sb.AppendLine();
string stNow = statusOf(inst);
if (validateError == null && validateResult is bool && (bool)validateResult)
    sb.AppendLine("校验通过 —— 可以考虑用 `InvokeUi=false` 调 `Execute()` 真正执行。");
else
    sb.AppendLine("校验未通过，_status = `" + stNow + "`。按 Status 语义处理：");
sb.AppendLine();
sb.AppendLine("- `ComponentNotInSysroot` → 组件当前存储不在库根内，先 SaveToLibrary 落到库内再试");
sb.AppendLine("- `ComponentHasNoPrototype` → 组件缺原型，需要先 Define Component Type 或带上原型");
sb.AppendLine("- `CannotOpenForModeling` → 组件当前不可建模");
sb.AppendLine("- `EquipmentHasSubComponentWithNoPrototype` → 设备下存在无原型的子组件");
sb.AppendLine();
sb.AppendLine("说明：本预演只调用了校验与路径计算，没有创建原型/组件，也没有写任何文件。");

return sb.ToString();
